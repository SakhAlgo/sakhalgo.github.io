Для расширения JSON новыми тестами — просто добавляйте объекты в массив questions, соблюдая структуру:

json
{
  "id": 21,
  "category": "Новая категория",
  "difficulty": "easy|medium|hard",
  "question": "Текст вопроса?",
  "options": ["Вариант A", "Вариант B", "Вариант C", "Вариант D"],
  "correct": 0,
  "explanation": "Пояснение правильного ответа (опционально)"
}
